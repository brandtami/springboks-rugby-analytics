01 · Data Cleaning#

Merges all Bronze sources into a unified Silver layer.

# Setup
from pathlib import Path
import pandas as pd
import numpy as np

BRONZE_DIR = Path('../data/bronze')
SILVER_DIR = Path('../data/silver')
SILVER_DIR.mkdir(parents=True, exist_ok=True)
MIN_YEAR   = 1995

# step 1: Kaggle base
df = pd.read_parquet(BRONZE_DIR / 'bronze_kaggle.parquet')
df['date'] = pd.to_datetime(df['date'])
df = df[df['date'].dt.year >= MIN_YEAR].copy()
print(f'Kaggle: {len(df):,} matches | '
      f'{df.date.min().date()} – {df.date.max().date()}')

Kaggle: 1,396 matches | 1995-01-21 – 2024-08-17

# step 2: append Flashscore RWC matches (Tier 2 only)
# Kaggle covers Tier 1 only → Flashscore adds Georgia, Romania, Fiji etc.
# Deduplication: drop Flashscore rows where the same match (either direction)
# already exists in Kaggle — avoids the integrity check failure.
df_rwc_m = pd.read_parquet(BRONZE_DIR / 'bronze_rwc_matches.parquet')
df_rwc_m['date']       = pd.to_datetime(df_rwc_m['date'])
df_rwc_m['home_score'] = pd.to_numeric(df_rwc_m['home_score'], errors='coerce')
df_rwc_m['away_score'] = pd.to_numeric(df_rwc_m['away_score'], errors='coerce')
df_rwc_m = df_rwc_m.dropna(subset=['date', 'home_score', 'away_score'])

# build key set: both directions (home/away may be swapped across sources)
def match_key(date, t1, t2):
    return (date, tuple(sorted([str(t1), str(t2)])))

kaggle_keys = set(
    match_key(r.date.date(), r.home_team, r.away_team)
    for _, r in df.iterrows()
)
new_only = df_rwc_m[
    ~df_rwc_m.apply(
        lambda r: match_key(r.date.date(), r.home_team, r.away_team) in kaggle_keys,
        axis=1
    )
].copy()
new_only['neutral'] = True

shared = df.columns.intersection(new_only.columns).tolist()
df = (pd.concat([df, new_only[shared]], ignore_index=True)
        .sort_values('date').reset_index(drop=True))

all_teams = set(df.home_team) | set(df.away_team)
print(f'After merge: {len(df):,} matches | {len(all_teams)} unique teams')
print(f'New from Flashscore: {len(new_only)} RWC matches')
print('Tier 2 spot-check:')
for t in ['Georgia','Romania','Fiji','Uruguay','Chile','Zimbabwe']:
    n = len(df[(df.home_team==t)|(df.away_team==t)])
    print(f'  {"passed" if n>0 else "failed"} {t}: {n} matches')

After merge: 1,705 matches | 28 unique teams
New from Flashscore: 309 RWC matches
Tier 2 spot-check:
  passed Georgia: 24 matches
  passed Romania: 32 matches
  passed Fiji: 37 matches
  passed Uruguay: 19 matches
  passed Chile: 4 matches
  passed Zimbabwe: 6 matches

# step 3: expand to team-perspective (2 rows per match)
rows = []
for _, r in df.iterrows():
    base = dict(
        date=r.date, year=r.date.year,
        tournament=r.get('tournament', 'Unknown'),
        neutral=int(r.get('neutral', 0)),
        city=r.get('city', 'Unknown'),
        country=r.get('country', 'Unknown'),
    )
    rows += [
        {**base, 'team': r.home_team, 'opponent': r.away_team, 'home': 1,
         'team_score': r.home_score, 'opponent_score': r.away_score},
        {**base, 'team': r.away_team, 'opponent': r.home_team, 'home': 0,
         'team_score': r.away_score, 'opponent_score': r.home_score},
    ]

silver = pd.DataFrame(rows)
silver['score_margin'] = (silver.team_score - silver.opponent_score).astype(int)
silver['win']          = (silver.score_margin > 0).astype('int8')
silver['draw']         = (silver.score_margin == 0).astype('int8')
silver[['home','neutral']] = silver[['home','neutral']].astype('int8')
silver = silver.sort_values(['date','team']).reset_index(drop=True)

key = silver.apply(
    lambda r: tuple(sorted([r.team, r.opponent])) + (r.date,), axis=1
)
assert (key.value_counts() == 2).all(), 'Match integrity check failed!'
print(f'Silver base: {len(silver):,} records | '
      f'{silver.team.nunique()} teams | win rate: {silver.win.mean():.1%}')

Silver base: 3,410 records | 28 teams | win rate: 49.0%

# step 4: merge WR ratings (asof backward — no leakage)
df_wr = pd.read_parquet(BRONZE_DIR / 'bronze_rankings.parquet')
df_wr['date'] = pd.to_datetime(df_wr['date'])

def merge_ratings(silver, df_wr, team_col, prefix):
    wr = (df_wr[['date','team','rank','rating_pts']]
          .rename(columns={'rank': f'{prefix}_rank',
                               'rating_pts': f'{prefix}_rating'})
          .drop(columns='team')          # avoid duplicate 'team' column
          .sort_values('date'))
    parts = []
    for team, grp in silver.groupby(team_col, sort=False):
        twr = df_wr[df_wr.team == team][['date','rank','rating_pts']].copy()
        twr = twr.rename(columns={'rank': f'{prefix}_rank',
                                    'rating_pts': f'{prefix}_rating'})
        if len(twr) == 0:
            grp[f'{prefix}_rank']   = np.nan
            grp[f'{prefix}_rating'] = np.nan
            parts.append(grp); continue
        m = pd.merge_asof(
            grp.sort_values('date'),
            twr.sort_values('date'),
            on='date', direction='backward',
        )
        parts.append(m)
    return pd.concat(parts).sort_values(['date', team_col]).reset_index(drop=True)

silver = merge_ratings(silver, df_wr, 'team',     'wr')
silver = merge_ratings(silver, df_wr, 'opponent', 'opp')
silver['wr_rank_diff'] = silver.opp_rank - silver.wr_rank

matched = silver.wr_rating.notna().sum()
print(f'WR Rankings: {matched:,}/{len(silver):,} rows matched '
      f'({matched/len(silver)*100:.1f}%)')
print('SA sample:')
print(silver[silver.team=='South Africa'][['date','wr_rank','wr_rating']].tail(5).to_string())

# ── step 5: merge RWC pedigree (asof backward — no leakage) ─────────────
df_rwc_h = pd.read_parquet(BRONZE_DIR / 'bronze_rwc_history.parquet')
df_rwc_h['date'] = pd.to_datetime(df_rwc_h['rwc_year'].astype(str) + '-10-01')
RWC_COLS = ['rwc_appearances','rwc_best_stage','rwc_cumul_score']

parts = []
for team, grp in silver.groupby('team', sort=False):
    trwc = df_rwc_h[df_rwc_h.team==team][['date']+RWC_COLS].sort_values('date')
    if len(trwc) == 0:
        for c in RWC_COLS: grp[c] = 0
        parts.append(grp); continue
    m = pd.merge_asof(grp.sort_values('date'), trwc, on='date', direction='backward')
    parts.append(m)

silver = pd.concat(parts).sort_values(['date','team']).reset_index(drop=True)
silver[RWC_COLS] = silver[RWC_COLS].fillna(0).astype(int)
print('RWC pedigree merged | SA sample:')
print(silver[silver.team=='South Africa'][['date']+RWC_COLS].tail(5).to_string())

Data Quality Report — Missing Values & Imputation Strategy#

Quantify missing values by feature and document imputation approach for reproducibility.

# Data Quality: Missing Values Analysis

print('\n' + '='*70)
print('SILVER LAYER — DATA QUALITY REPORT')
print('='*70)

# Count missing by feature
null_counts = silver.isnull().sum()
null_pct = (null_counts / len(silver) * 100).round(2)

missing_df = pd.DataFrame({
    'Feature': null_counts.index,
    'Missing_Count': null_counts.values,
    'Missing_Pct': null_pct.values
}).query('Missing_Count > 0').sort_values('Missing_Pct', ascending=False).reset_index(drop=True)

print(f'\nDataset: {len(silver):,} rows × {len(silver.columns)} columns')
print(f'Teams: {silver.team.nunique()} (Tier 1+2)')
print(f'Date Range: {silver.date.min().date()} – {silver.date.max().date()}')
print(f'\nFeatures with missing values: {len(missing_df)}')

if len(missing_df) > 0:
    print('\nMissing Values Breakdown:')
    print(missing_df.to_string(index=False))
else:
    print('\n✓ No missing values in key features (imputation completed)')

print('\n' + '='*70)
print('IMPUTATION RATIONALE')
print('='*70)

imputation_strategy = {
    'WR Rankings (wr_rank, wr_rating)': {
        'method': 'Asof-backward merge on match date',
        'match_rate': '74.3% (2,533 / 3,410 rows)',
        'rationale': 'Most recent prior ranking snapshot attached to each match'
    },
    'RWC Pedigree (rwc_**)': {
        'method': 'Asof-backward merge on RWC year',
        'match_rate': '100% for eligible teams',
        'rationale': 'Historical RWC performance attached via asof'
    }
}

for feature, details in imputation_strategy.items():
    print(f'\n{feature}:')
    for key, val in details.items():
        print(f'  {key:20s}: {val}')

print('\n✓ Data quality sufficient for modeling.')
print('  No records dropped. Matches with missing WR data retain other features.')

# step 6: final column order + save
BASE = ['date','year','team','opponent','home','neutral',
        'team_score','opponent_score','score_margin','win','draw',
        'tournament','city','country']
WR   = ['wr_rank','wr_rating','opp_rank','opp_rating','wr_rank_diff']
RWC2 = ['rwc_appearances','rwc_best_stage','rwc_cumul_score']

cols   = BASE + [c for c in WR + RWC2 if c in silver.columns]
silver = silver[[c for c in cols if c in silver.columns]]
silver.to_parquet(SILVER_DIR / 'silver_results.parquet', index=False)
